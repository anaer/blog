多仓库目录环境配置

采用 includeIf 指令实现智能切换：

ini
# ~/.gitconfig 主配置文件[includeIf "gitdir:D:/work/"]    path = .gitconfig-work  # 工作账号配置[includeIf "gitdir:D:/personal/"]    path = .gitconfig-personal  # 私人账号配置[includeIf "onbranch:main"]    path = .gitconfig-personal[includeIf "hasconfig:remote.*.url:https://github.com/**"]    path = .gitconfig-personal
